```sql
--prepare

SELECT step_id, COUNT(keyword_id)
FROM step
         NATURAL JOIN step_keyword
         NATURAL JOIN keyword
GROUP BY 1;


--prepare
SELECT step.step_id,
       keyword.keyword_id
FROM step
         CROSS JOIN
     keyword;


--solution-1

SELECT step_id, keyword_id
FROM step
         CROSS JOIN keyword
WHERE step_name ~ ('\y' || keyword_name || '\y');
-- WHERE step_name ~ concat('\y', keyword_name, '\y')
-- WHERE step_name ~* concat('\y', keyword_name, '\y')

--solution-2

SELECT step_id, keyword_id, keyword_name, step_name
FROM step,
     keyword
WHERE step_name ~ keyword_name;
-- WHERE position(keyword_name in step_name)>0;

--solution-3

INSERT INTO step_keyword
SELECT step_id, keyword_id
FROM keyword
         JOIN step ON step_name LIKE CONCAT('% ', keyword_name, ' %')
    OR step_name LIKE CONCAT('% ', keyword_name, ',%')
    OR step_name LIKE CONCAT('% ', keyword_name, '(%')
    OR step_name LIKE CONCAT('% ', keyword_name);
```

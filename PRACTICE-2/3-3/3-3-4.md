# Задание


Вывести образовательные программы, для которых минимальный балл ЕГЭ по каждому предмету больше или равен 40 баллам. 
Программы вывести в отсортированном по алфавиту виде.



# Результат

+-------------------------------------+
| name_program                        |
+-------------------------------------+
| Мехатроника и робототехника         |
| Прикладная математика и информатика |
+-------------------------------------+


```sql
--solution-1

SELECT name_program
FROM program
         NATURAL JOIN program_subject
GROUP BY name_program
HAVING SUM((min_result >= 40)::INT) = 3
ORDER BY name_program;

--solution-2
SELECT name_program
FROM program
         NATURAL JOIN program_subject
GROUP BY name_program
HAVING MIN(min_result) >= 40
ORDER BY name_program;


--solution-3

SELECT name_program
FROM program
         JOIN (SELECT MIN(min_result), program_id
               FROM program_subject
               GROUP BY program_id
               HAVING MIN(min_result) >= 40) query_in USING (program_id)
GROUP BY name_program
ORDER BY 1;


--solution-4

SELECT name_program
FROM program
WHERE program_id NOT IN (SELECT program_id
                         FROM program_subject
                         WHERE min_result < 40)
ORDER BY 1;


--solution-5

SELECT name_program
FROM program
         INNER JOIN program_subject USING (program_id)
GROUP BY name_program
HAVING MIN(min_result) >= 40
ORDER BY name_program;
```

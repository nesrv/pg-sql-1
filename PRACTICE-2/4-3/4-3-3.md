# Задание

Для клиентов у которых сумма заказов выше средней по суммам заказов клиентов (общей стоимости всех заказов клиентов), вывести имя, общую сумму всех заказов, количество заказов, количество заказанных книг.

Этим клиентам мы предложим специальную программу лояльности!

Информацию отсортировать по имени клиентов ( в алфавитном порядке).

Результат

| name_client   | Общая*сумма*заказов | Заказов_всего | Книг_всего |
| ------------- | ------------------- | ------------- | ---------- |
| Абрамова Катя | 2131.49             | 1             | 4          |
| Баранов Павел | 2291.89             | 2             | 4          |

```sql

CREATE TABLE clients (
    client_id SERIAL PRIMARY KEY,
    name_client VARCHAR(100) NOT NULL,

SELECT
name_client,
sum(price * bb.amount) AS Общая_сумма_заказов,
sum(bb.amount) AS Книг_всего,
count(bb.buy_id) AS Заказов_всего

WITH t1 AS (
SELECT sum(price * bb.amount) AS bb
FROM buy_book bb
inner JOIN book b USING (book_id)
GROUP BY buy_id
)
SELECT name_client,
sum(price * bb.amount) AS Общая_сумма_заказов,
count(DISTINCT bb.buy_id) AS Заказов_всего,
sum(bb.amount) AS Книг_всего
FROM client
NATURAL JOIN buy
NATURAL JOIN buy_book bb
inner JOIN book b USING (book_id)
GROUP BY name_client
HAVING sum(price * bb.amount) > (SELECT avg(bb) FROM t1)
ORDER by name_client;


WITH order_totals AS (
    -- Calculate total amount for each order
    SELECT 
        buy_id,
        SUM(b.price * bb.amount) AS order_sum
    FROM buy_book bb
    JOIN book b USING (book_id)
    GROUP BY buy_id
),
client_stats AS (
    -- Calculate statistics for each client
    SELECT 
        c.name_client,
        SUM(b.price * bb.amount) AS total_order_amount,
        SUM(bb.amount) AS total_books,
        COUNT(DISTINCT bb.buy_id) AS total_orders
    FROM clients c
    JOIN buy USING (client_id)
    JOIN buy_book bb USING (buy_id)
    JOIN book b USING (book_id)
    GROUP BY c.name_client
)
-- Select clients with above average order totals
SELECT 
    name_client,
    total_order_amount AS Общая_сумма_заказов,
    total_books AS Книг_всего,
    total_orders AS Заказов_всего
FROM client_stats
WHERE total_order_amount > (
    SELECT AVG(order_sum) 
    FROM order_totals
)
ORDER BY name_client;
```

SELECT q.name_question,
       a.name_answer,
       CASE
           WHEN a.is_correct = 1 THEN 'Верно'
           ELSE 'Неверно'
           END AS "Результат"
FROM testing t
         JOIN question q USING (question_id)
         JOIN answer a USING (answer_id)
WHERE t.attempt_id = 7;

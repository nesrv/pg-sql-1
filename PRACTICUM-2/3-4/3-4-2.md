# Задание

Из таблицы applicant, созданной на предыдущем шаге, удалить записи, 
если абитуриент на выбранную образовательную программу не набрал минимального балла
хотя бы по одному предмету (использовать запрос из предыдущего урока).

```sql
--solution-1

DELETE
FROM applicant
    USING program_subject, enrollee_subject
WHERE applicant.program_id = program_subject.program_id
  AND applicant.enrollee_id = enrollee_subject.enrollee_id
  AND enrollee_subject.subject_id = program_subject.subject_id
  AND enrollee_subject.result < program_subject.min_result;


--solution-2

DELETE
FROM applicant
    USING applicant
        NATURAL JOIN program_subject
        NATURAL JOIN enrollee_subject
WHERE result < min_result;

DELETE
FROM applicant
    USING program_subject
        NATURAL JOIN enrollee_subject
WHERE result < min_result;
```

# Задание

Вывести жанр(ы), в котором было заказано меньше всего экземпляров книг, указать это количество.
Учитывать только жанры, в которых была заказана хотя бы одна книга.

При реализации в основном запросе не используйте LIMIT, поскольку жанров с минимальным количеством
заказанных книг может быть несколько.

Результат:

| name_genre | Количество |
| ---------- | ---------- |
| Поэзия     | 4          |

```sql
--исходные данные
CREATE TABLE buy_book (
    buy_book_id SERIAL PRIMARY KEY,
    buy_id      INTEGER,
    book_id     INTEGER,
    amount      INTEGER,
    FOREIGN KEY (book_id) REFERENCES book (book_id) ON DELETE CASCADE
);


INSERT INTO buy_book (buy_id, book_id, amount)
VALUES
    (1, 1, 1),
    (1, 7, 2),
    (1, 3, 1),
    (2, 8, 2),
    (3, 3, 2),
    (3, 2, 1),
    (3, 1, 1),
    (4, 5, 1);

```

```sql
--prepare

SELECT	
	name_genre,
	sum(buy_book.amount) AS Количество
FROM
	book
NATURAL JOIN genre
INNER JOIN buy_book
		USING (book_id)
GROUP BY
	name_genre
HAVING
	sum(buy_book.amount) = 4



SELECT name_genre, sum(buy_book.amount) AS Количество
FROM book
NATURAL JOIN genre
INNER JOIN buy_book
	USING (book_id)
GROUP BY
	name_genre
HAVING
	COUNT(buy_book.amount) IN (
        SELECT MIN(amount)
FROM buy_book
GROUP BY 
	book_id);




WITH GenreAmounts AS (
SELECT name_genre, SUM(buy_book.amount) AS Количество
FROM genre
JOIN book
	USING(genre_id)
JOIN buy_book
	USING(book_id)
GROUP BY
	name_genre
)
SELECT name_genre, Количество
FROM GenreAmounts
WHERE Количество = (
	SELECT MIN(Количество)
FROM GenreAmounts);

```

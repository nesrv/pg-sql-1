# Задание

В последний заказ (таблица buy_book) клиента Баранов Павел добавить по одному экземпляру всех книг Достоевского, которые есть в таблице book.

```sql

SELECT name_client, buy_book_id, buy_id , book_id, amount  FROM clients
NATURAL JOIN buy
NATURAL JOIN buy_book
WHERE name_client LIKE '%Баранов%'




INSERT INTO buy_book(buy_id, book_id, amount)
SELECT 
(SELECT max(buy_id) AS buy_id
FROM buy
NATURAL JOIN client
WHERE name_client LIKE '%Баранов%'
), 
book_id, 1 FROM  book
NATURAL JOIN author
WHERE name_author LIKE '%Достоевский%';


SELECT * FROM buy_book;


SELECT MAX(buy_id), book_id, 1
FROM buy
NATURAL JOIN clients
NATURAL JOIN book
NATURAL JOIN author
WHERE name_client LIKE '%Баранов%' AND name_author LIKE '%Достоевский%'
GROUP BY 2;
```

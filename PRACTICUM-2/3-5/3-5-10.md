SELECT student,

-- solution-1

WITH get_time_lesson(student_name, lesson, max_submission_time)
         AS (SELECT student_name, CONCAT(module_id, '.', lesson_position), MAX(submission_time)
             FROM step_student
                      INNER JOIN step USING (step_id)
                      INNER JOIN lesson USING (lesson_id)
                      INNER JOIN student USING (student_id)
             WHERE result = 'correct'
             GROUP BY 1, 2
             ORDER BY 1),
     get_students(student_name)
         AS (SELECT student_name
             FROM get_time_lesson
             GROUP BY student_name
             HAVING COUNT(lesson) = 3)
SELECT student_name                                 AS Студент,
       lesson                                       AS Урок,
       TO_TIMESTAMP(max_submission_time)::timestamp AS Макс_время_отправки,
       COALESCE(
               CEIL((max_submission_time -
                     LAG(max_submission_time) OVER (PARTITION BY student_name ORDER BY max_submission_time))::numeric /
                    86400
               )::text,
               '-'
       )                                            AS Интервал
FROM get_time_lesson
WHERE student_name IN (SELECT * FROM get_students)
ORDER BY 1, 3;


-- solution-2

WITH submitted_time(stud, les, time) AS
         (SELECT student_name, CONCAT(module_id, '.', lesson_position), MAX(submission_time)
          FROM student
                   NATURAL JOIN step_student
                   NATURAL JOIN step
                   NATURAL JOIN lesson
          WHERE result = 'correct'
          GROUP BY 1, 2),
     filtered_students AS (SELECT stud
                           FROM submitted_time
                           GROUP BY 1
                           HAVING COUNT(DISTINCT les) = 3)
SELECT stud                          AS Студент,
       les                           AS Урок,
       TO_TIMESTAMP(time)::timestamp AS Макс_время_отправки,
       COALESCE(
               CEIL((time - LAG(time) OVER (PARTITION BY stud ORDER BY time))::numeric /
                    86400
               )::text,
               '-'
       )                             AS Интервал
FROM submitted_time
         NATURAL JOIN filtered_students;
